Koppel je eigen Python Script of Discord Bot

Werk met je hele klas of groepje samen aan de AI achterkant.

easymail_bot.py
Python 3.10+ 
import google.generativeai as genai

# Configureer je Gemini API Key
genai.configure(api_key="JOUW_API_KEY")

def maak_email(onderwerp, toon="zakelijk"):
    model = genai.GenerativeModel('gemini-3-flash-preview')
    prompt = f"Schrijf een {toon} e-mail over: {onderwerp}"
    
    response = model.generate_content(prompt)
    return response.text

# Test het script
print(maak_email("Afwezig wegens ziekte"))
